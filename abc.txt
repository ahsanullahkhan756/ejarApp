import React, { useState, useEffect } from 'react';
import { View, TouchableOpacity, Image, Alert, Linking, Platform } from 'react-native';
import { useSelector, useDispatch } from 'react-redux';
import { useTranslation } from 'react-i18next';
import ImagePicker from 'react-native-image-crop-picker';
import moment from 'moment';
import { PermissionsAndroid } from 'react-native';
import { COMMON_TEXT, EJAR } from '../../constants';
import { theme, commonStyles, styles } from '../../theme';
import { Typography } from '../../components';

const InformationIds = ({ onValidate, setCurrentStep }: any) => {
  const user = useSelector((state) => state?.user?.userDetails);
  const [hasValidated, setValidated] = useState(new Array(3).fill(true));
  const dispatch = useDispatch();
  const { t } = useTranslation();

  // Initialize form fields
  const [idCardNumber, setIdCardNumber] = useState(user?.idCardNumber ?? "");
  const [frontImage, setFrontImage] = useState(user?.idcardPicture?.base64 ?? null);
  const [backImage, setBackImage] = useState(user?.idcardPictureBack?.base64 ?? null);
  const [isTakingFront, setIsTakingFront] = useState(true);

  const showPermissionDeniedAlert = () => {
    Alert.alert(
      t(COMMON_TEXT.CAMERA_PERMISSION_REQUIRED),
      t(COMMON_TEXT.CAMERA_PERMISSION_MESSAGE),
      [
        {
          text: t(COMMON_TEXT.CANCEL),
          style: "cancel"
        },
        { 
          text: t(COMMON_TEXT.OPEN_SETTINGS), 
          onPress: () => Linking.openSettings()
        }
      ]
    );
  };

  const requestCameraPermission = async () => {
    if (Platform.OS === 'android') {
      try {
        const granted = await PermissionsAndroid.request(
          PermissionsAndroid.PERMISSIONS.CAMERA,
          {
            title: t(COMMON_TEXT.CAMERA_PERMISSION_TITLE),
            message: t(COMMON_TEXT.CAMERA_PERMISSION_MESSAGE),
            buttonPositive: t(COMMON_TEXT.OK)
          }
        );
        
        if (granted === PermissionsAndroid.RESULTS.GRANTED) {
          openCamera();
        } else {
          showPermissionDeniedAlert();
        }
      } catch (err) {
        showPermissionDeniedAlert();
      }
    } else {
      // iOS handles permissions through ImagePicker directly
      openCamera();
    }
  };

  const openCamera = () => {
    ImagePicker.openCamera({
      width: 300,
      height: 400,
      cropping: true,
      includeBase64: true,
      forceJpg: true,
    })
    .then((image) => {
      if (isTakingFront) {
        setFrontImage(image.data); // Using base64 data
      } else {
        setBackImage(image.data);
      }
      setIsTakingFront(false);
    })
    .catch((error) => {
      if (error.code === 'E_PERMISSION_MISSING') {
        showPermissionDeniedAlert();
      }
      // Other errors are ignored (like when user cancels)
    });
  };

  const removeImage = (isFront: boolean) => {
    if (isFront) {
      setFrontImage(null);
      setIsTakingFront(true);
    } else {
      setBackImage(null);
    }
  };

  return (
    <View style={{}}>
      <View style={commonStyles.lineBar} />
      <Typography textType="bold" align="center" size={theme.fontSize.large24}>
        {EJAR.ID_CARD_INFORMATION}
      </Typography>

      {(!frontImage || !backImage) && (
        <TouchableOpacity onPress={requestCameraPermission}>
          <View row style={styles.button}>
            <Image
              source={IMAGES.cameraIcon}
              style={styles.cameraIcon2}
              resizeMode="contain"
            />
            <Typography>
              {isTakingFront
                ? COMMON_TEXT.TAKE_FRONT_PICTURE
                : COMMON_TEXT.TAKE_BACK_PICTURE}
            </Typography>
          </View>
        </TouchableOpacity>
      )}

      {frontImage && (
        <View>
          <Image
            source={{ uri: `data:image/jpeg;base64,${frontImage}` }}
            style={styles.imagePreview}
            resizeMode="cover"
          />
          <TouchableOpacity
            style={styles.deleteIcon}
            onPress={() => removeImage(true)}
          >
            <Image source={IMAGES.cross} style={styles.deleteIconImg} />
          </TouchableOpacity>
        </View>
      )}

      {backImage && (
        <View>
          <Image
            source={{ uri: `data:image/jpeg;base64,${backImage}` }}
            style={styles.imagePreview}
            resizeMode="cover"
          />
          <TouchableOpacity
            style={styles.deleteIcon}
            onPress={() => removeImage(false)}
          >
            <Image source={IMAGES.cross} style={styles.deleteIconImg} />
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
};

export default InformationIds;
